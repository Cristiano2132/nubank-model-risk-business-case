% --- Dynamic Spotlight / Fade Commands ---
% Active overlay step -> vivid colors; Inactive overlay steps -> dimmed/faded colors
\newcommand{\spotitem}[4]{%
  % #1: overlay step (e.g. 1)
  % #2: bullet color when active
  % #3: prefix/lead (bold with colon)
  % #4: description text
  \alt<#1>{%
    \par\hangindent=1.1em\hangafter=1\noindent%
    \makebox[1.1em][l]{\textcolor{#2}{\scriptsize$\bullet$}}%
    \textbf{\textcolor{textPrimary}{#3:}} \textcolor{textPrimary}{#4}\par
  }{%
    \par\hangindent=1.1em\hangafter=1\noindent%
    \makebox[1.1em][l]{\textcolor{bulletDimmed}{\scriptsize$\bullet$}}%
    \textbf{\textcolor{textDimmed}{#3:}} \textcolor{textDimmed}{#4}\par
  }%
  \vspace{0.8mm}
}

\newcommand{\spotitemplain}[4]{%
  % #1: overlay step
  % #2: bullet color when active
  % #3: prefix/lead (bold without colon)
  % #4: description text
  \alt<#1>{%
    \par\hangindent=1.1em\hangafter=1\noindent%
    \makebox[1.1em][l]{\textcolor{#2}{\scriptsize$\bullet$}}%
    \textbf{\textcolor{textPrimary}{#3}} \textcolor{textPrimary}{#4}\par
  }{%
    \par\hangindent=1.1em\hangafter=1\noindent%
    \makebox[1.1em][l]{\textcolor{bulletDimmed}{\scriptsize$\bullet$}}%
    \textbf{\textcolor{textDimmed}{#3}} \textcolor{textDimmed}{#4}\par
  }%
  \vspace{0.8mm}
}

\newcommand{\spotsubitem}[3]{%
  % Sub-bullet with indentation
  \alt<#1>{%
    \par\hangindent=1.8em\hangafter=1\noindent\hspace*{0.9em}%
    \makebox[0.9em][l]{\textcolor{#2}{\scriptsize--}}%
    \textcolor{textPrimary}{#3}\par
  }{%
    \par\hangindent=1.8em\hangafter=1\noindent\hspace*{0.9em}%
    \makebox[0.9em][l]{\textcolor{bulletDimmed}{\scriptsize--}}%
    \textcolor{textDimmed}{#3}\par
  }%
  \vspace{0.6mm}
}
